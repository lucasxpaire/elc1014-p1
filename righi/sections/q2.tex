\section{Questao 2 - Greedy Best-First Search e A*}

\subsection{Formulacao do problema em espaco de estados}
\[
P = \langle Z, \mathcal{A}, T, z_0, G, c, h \rangle
\]
\begin{itemize}[leftmargin=1.5em]
  \item $Z$: conjunto de estados (nos do grafo), $Z=\{A,B,C,\dots,T\}$.
  \item $z_0=A$: estado inicial.
  \item Objetivo: alcancar o estado $T$.
  \item $T(\cdot)$: funcao de transicao segundo as arestas dirigidas do enunciado.
  \item $c(z,a,z')$: custo da aresta percorrida.
  \item $h(z)$: heuristica fornecida no enunciado (estimativa ate $T$).
\end{itemize}

\subsection{Comparacao final: Greedy vs A*}
\textbf{Notacao:}
\[
b = \text{fator de ramificacao},\quad d = \text{profundidade da solucao},\quad m = \text{profundidade maxima}.
\]

\begin{enumerate}[label=\alph*)]
  \item \textbf{Completude}: Greedy nao e completo em geral; A* e completo em condicoes usuais (custos positivos e espaco finito/adequado).
  \item \textbf{Otimalidade}: Greedy nao garante otimo; A* garante otimo com heuristica admissivel/consistente.
  \item \textbf{Consumo de memoria}: ambos guardam fronteira ordenada; A* tende a manter mais estados por considerar custo acumulado.
  \item \textbf{Dependencia da heuristica}: Greedy e altamente dependente; A* tambem depende, mas de forma mais robusta.
  \item \textbf{Custo computacional}: Greedy costuma expandir menos quando a heuristica aponta bem; A* expande mais, mas com melhor garantia.
  \item \textbf{Qualidade da solucao}: A* superior em garantia de qualidade; Greedy pode ser bom, mas pode errar caminho.
  \item \textbf{Comportamento em grafos grandes}: Greedy pode ser rapido, porem arriscado; A* e mais confiavel, mas pode consumir muita memoria.
\end{enumerate}
